cmake_minimum_required(VERSION 3.24)
project(shante_core VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SHANTE_SYSTEM_ZSTD "Use a package-manager supplied Zstandard" OFF)
option(SHANTE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
if(SHANTE_SYSTEM_ZSTD)
  find_package(zstd 1.5.7 CONFIG REQUIRED)
else()
  include(FetchContent)
  set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_DICTBUILDER OFF CACHE BOOL "" FORCE)
  set(ZSTD_MULTITHREAD_SUPPORT ON CACHE BOOL "" FORCE)
  FetchContent_Declare(zstd
    URL https://github.com/facebook/zstd/releases/download/v1.5.7/zstd-1.5.7.tar.gz
    URL_HASH SHA256=eb33e51f49a15e023950cd7825ca74a4a2b43db8354825ac24fc1b7ee09e6fa3
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR build/cmake)
  FetchContent_MakeAvailable(zstd)
endif()
if(TARGET zstd::libzstd_static)
  set(SHANTE_ZSTD zstd::libzstd_static)
else()
  set(SHANTE_ZSTD libzstd_static)
endif()
add_library(shante_kernel src/kernel.cpp)
target_include_directories(shante_kernel PUBLIC include)
target_link_libraries(shante_kernel PRIVATE ${SHANTE_ZSTD})
target_compile_definitions(shante_kernel PRIVATE _FILE_OFFSET_BITS=64)
add_executable(shante-core src/main.cpp)
target_link_libraries(shante-core PRIVATE shante_kernel)
if(MINGW)
  target_link_options(shante-core PRIVATE -municode -static)
endif()
foreach(target shante_kernel shante-core)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    if(SHANTE_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME native_contract COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:shante-core>)
  set_tests_properties(native_contract PROPERTIES TIMEOUT 120)
endif()
install(TARGETS shante-core RUNTIME DESTINATION bin COMPONENT ShanteCore)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE" DESTINATION licenses
  RENAME HAMMA-LICENSE.txt COMPONENT ShanteCore)
if(NOT SHANTE_SYSTEM_ZSTD)
  install(FILES "${zstd_SOURCE_DIR}/LICENSE" DESTINATION licenses
    RENAME Zstandard-LICENSE.txt COMPONENT ShanteCore)
endif()
